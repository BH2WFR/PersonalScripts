cmake_minimum_required(VERSION 3.21)
project(locale_launcher LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(DETOURS_SOURCE "" CACHE PATH "Microsoft Detours v4.0.1 source directory")
if(NOT MSVC OR NOT EXISTS "${DETOURS_SOURCE}/src/detours.h")
  message(FATAL_ERROR "Use MSVC and set DETOURS_SOURCE to a Microsoft Detours v4.0.1 checkout.")
endif()
set(detours_units detours modules disasm image creatwth disolx86 disolx64 disolia64 disolarm disolarm64)
add_library(detours STATIC)
foreach(unit IN LISTS detours_units)
  target_sources(detours PRIVATE "${DETOURS_SOURCE}/src/${unit}.cpp")
endforeach()
target_include_directories(detours PUBLIC "${DETOURS_SOURCE}/src")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_executable(locale-start WIN32 start.cpp)
target_link_libraries(locale-start PRIVATE detours shell32)
add_library(locale-settings SHARED settings.cpp settings.def)
target_link_libraries(locale-settings PRIVATE detours user32 gdi32)
install(TARGETS locale-start locale-settings RUNTIME DESTINATION .)
if(CMAKE_SIZEOF_VOID_P EQUAL 4)
  add_executable(locale-info WIN32 locale-info.cpp)
  target_link_libraries(locale-info PRIVATE user32 advapi32)
  set_target_properties(locale-info PROPERTIES OUTPUT_NAME "_locale-info" SUFFIX ".com")
endif()
